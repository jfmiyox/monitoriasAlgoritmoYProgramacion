\begin{analisis}{Diccionario Traductor de Enunciados}
\scriptsize
\begin{tabularx}{\linewidth}{>{\bfseries\color{AzulMarino}}p{3.2cm} >{\raggedright\arraybackslash}X}
\toprule
\textbf{Frase del parcial} & \textbf{Cómo se traduce} \\
\midrule
``Al menos X'' / ``mínimo X'' & \codigo{dato >= X} \\
``Supera X'' / ``más de X'' & \codigo{dato > X} \\
``A lo sumo X'' / ``máximo X'' & \codigo{dato <= X} \\
``Entre A y B inclusive'' & \codigo{A <= x \&\& x <= B} \\
``Estrictamente entre A y B'' & \codigo{A < x \&\& x < B} \\
``Múltiplo / divisible por k'' & \codigo{x \% k == 0} \\
``Para N estudiantes...'' & Ciclo \codigo{for (0..N-1)} \\
``Hasta ingresar 0 / -999'' & Ciclo \codigo{while} con centinela \\
``Validar / mostrar menú'' & Ciclo \codigo{do-while} \\
``Guardar listado / tabla'' & Vector \codigo{v[MAX]} / Matriz \codigo{m[][MAX\_C]} \\
\bottomrule
\end{tabularx}
\end{analisis}

\columnbreak

\begin{consejo}{Selector Rápido: ¿Qué ciclo uso?}
\scriptsize
\begin{itemize}[leftmargin=10pt,itemsep=2pt]
  \item \textbf{Sé exactamente cuántas veces} se repite ($N$, 10 notas, filas/columnas) $\Rightarrow$ \codigo{for}.
  \item \textbf{No sé cuántas veces} (termina con centinela o condición que puede ser falsa desde el inicio) $\Rightarrow$ \codigo{while}.
  \item \textbf{Debe ocurrir al menos 1 vez} (menús, validar entrada en rango) $\Rightarrow$ \codigo{do-while}.
\end{itemize}
\end{consejo}

\vspace{-4pt}
\begin{ejercicio}{Checklist Anti-Pánico (Últimos 5 Minutos)}
\scriptsize
\begin{enumerate}[leftmargin=11pt,itemsep=1pt]
  \item Contadores y acumuladores inicializados en \codigo{0}.
  \item División real con \codigo{5.0} o \codigo{static\_cast<double>}.
  \item Condiciones con \codigo{==} (no \codigo{=}) y rangos con \codigo{\&\&}.
  \item Todo \codigo{case} lleva \codigo{break;} y existe \codigo{default:}.
  \item Arreglos fijos con \codigo{MAX}, índices \codigo{0..n-1} y matrices \codigo{[][MAX\_C]}.
  \item Mínimo/máximo inicializados con primer dato válido, no \codigo{0}.
  \item Prueba de escritorio ejecutada con caso normal y caso límite.
\end{enumerate}
\end{ejercicio}
